// D-24 的證據樣式自測樣本(規則包自檢 check.py 的 situation_sample_problems 讀;格式見那支檔〈證據樣式自測〉)
// === main-ok 在 execute 區塊內讀完串流
suspend fun download(client: HttpClient, url: String, stream: OutputStream) {
    client.prepareGet(url).execute { httpResponse ->
        val channel: ByteReadChannel = httpResponse.body()
        while (!channel.exhausted()) {
            val chunk = channel.readRemaining(8192)
            chunk.transferTo(stream)
        }
    }
}
// === bad API-30 把 response 帶出 execute 才讀
suspend fun download(client: HttpClient, url: String): ByteReadChannel {
    val response = client.prepareGet(url).execute()
    return response.body()
}
