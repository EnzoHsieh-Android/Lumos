// D-9 的證據樣式自測樣本(規則包自檢 check.py 的 situation_sample_problems 讀;格式見那支檔〈證據樣式自測〉)
// === main-ok 標準寫法:supervisorScope 裡每支 runCatching
suspend fun load(api: Api): Ui = supervisorScope {
    val a = async { api.a() }
    val b = async { api.b() }
    Ui(runCatching { a.await() }, runCatching { b.await() })
}
// === main-ok 每個子協程自己 try/catch
suspend fun syncAll(api: Api) = supervisorScope {
    launch { try { api.a() } catch (e: IOException) { log(e) } }
    launch { try { api.b() } catch (e: IOException) { log(e) } }
}
// === bad CO-11 supervisorScope 子協程沒自己處理例外
suspend fun syncAll(api: Api) = supervisorScope {
    launch { api.a() }
    launch { api.b() }
}
// === bad API-4 supervisorScope 裡 await 沒包
suspend fun load(api: Api): Ui = supervisorScope {
    val a = async { api.a() }
    Ui(a.await())
}
