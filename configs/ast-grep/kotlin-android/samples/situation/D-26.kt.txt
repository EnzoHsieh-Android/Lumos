// D-26 的證據樣式自測樣本(規則包自檢 check.py 的 situation_sample_problems 讀;格式見那支檔〈證據樣式自測〉)
// === main-ok Flow 可能一筆都不發:firstOrNull
suspend fun find(repo: Repo): Item? = repo.find().firstOrNull()
// === main-ok Channel 只要第一個:consume 結束時取消 channel
suspend fun <E> ReceiveChannel<E>.consumeFirst(): E = consume { return receive() }
// === alt-ok 多個消費者分食同一個 channel:for (msg in channel)
suspend fun worker(channel: ReceiveChannel<Msg>) {
    for (msg in channel) handle(msg)
}
// === bad FLOW-29 produce 建的 Channel 只 receive 一個就丟
suspend fun firstTick(scope: CoroutineScope): Int {
    val ch = scope.produce { while (true) send(tick()) }
    return ch.receive()
}
// === bad fan-out 用 consumeEach(結束時取消整個 channel)
suspend fun worker(channel: ReceiveChannel<Msg>) {
    channel.consumeEach { handle(it) }
}
