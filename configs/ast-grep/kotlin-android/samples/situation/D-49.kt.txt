// === main-ok stateIn 的狀態先有收集者再斷言 value
@Test
fun score() = runTest {
    val vm = Vm(fake)
    backgroundScope.launch(UnconfinedTestDispatcher(testScheduler)) { vm.score.collect {} }
    fake.emit(1)
    assertEquals(1, vm.score.value)
}
// === main-ok 有掛起點先 advanceUntilIdle 再斷言
@Test
fun loads() = runTest {
    val vm = Vm(fake)
    vm.load()
    advanceUntilIdle()
    assertEquals(Ui.Done, vm.state.value)
}
// === alt-ok 能改 API:直接 await
@Test
fun loads() = runTest {
    val vm = Vm(fake)
    vm.loadAsync().await()
    assertEquals(Ui.Done, vm.state.value)
}
