// === main-ok Mutex 保護的快取
class NewsRepo(private val api: Api) {
    private val latestNewsMutex = Mutex()
    suspend fun getLatestNews(refresh: Boolean = false): List<Headline> {
        if (refresh || latestNews.isEmpty()) latestNewsMutex.withLock { this.latestNews = api.fetch() }
        return latestNews
    }
}
