// D-5 的證據樣式自測樣本(規則包自檢 check.py 的 situation_sample_problems 讀;格式見那支檔〈證據樣式自測〉)
// === main-ok 標準寫法:combine 再 stateIn(WhileSubscribed)
class ProfileVm(private val repo: Repo) : ViewModel() {
    val uiState: StateFlow<UiState> =
        combine(repo.user, repo.settings, query) { u, s, q -> UiState(u, s, q) }
            .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), UiState.Loading)
}
// === alt-ok 某來源只在第一次需要:first() 取一次再進 combine
class ProfileVm(private val repo: Repo) : ViewModel() {
    suspend fun build(): Flow<Ui> {
        val config = repo.config.first()
        return combine(repo.user, repo.settings) { u, s -> Ui(u, s, config) }
    }
}
// === bad STATE-4 共享策略用 Eagerly
class ProfileVm(private val repo: Repo) : ViewModel() {
    val uiState = combine(repo.user, repo.settings) { u, s -> UiState(u, s) }
        .stateIn(viewModelScope, SharingStarted.Eagerly, UiState.Loading)
}
