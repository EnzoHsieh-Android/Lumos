// D-16 的證據樣式自測樣本(規則包自檢 check.py 的 situation_sample_problems 讀;格式見那支檔〈證據樣式自測〉)
// === main-ok 只要最新值:conflate
fun Flow<Int>.latestOnly() = conflate()
// === main-ok 只要最新值:collectLatest
suspend fun render(src: Flow<Frame>) {
    src.collectLatest { draw(it) }
}
// === main-ok 要保留但有上限:buffer(64, DROP_OLDEST)
fun Flow<Int>.bounded() = buffer(64, onBufferOverflow = BufferOverflow.DROP_OLDEST)
// === main-ok 回呼轉 Flow:trySend 有接結果
fun events(src: Source) = callbackFlow {
    val l = Listener { v -> trySend(v).onFailure { log(it) } }
    src.add(l)
    awaitClose { src.remove(l) }
}
// === main-ok Channel 用有上限的容量
class Pipe {
    val ch = Channel<Int>(64)
}
// === bad FLOW-30 無上限的緩衝(代碼審 r1 接手 F5)
fun Flow<Int>.unbounded() = buffer(Channel.UNLIMITED)
// === bad FLOW-30 無上限的 Channel
class Pipe {
    val ch = Channel<Int>(Channel.UNLIMITED)
}
// === bad FLOW-31 callbackFlow 裡忽略 trySend 結果(代碼審 r1 接手 F5)

    val l = Listener { v -> trySend(v) }
    src.add(l)
    awaitClose { src.remove(l) }
}
