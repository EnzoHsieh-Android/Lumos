// D-1 的證據樣式自測樣本(規則包自檢 check.py 的 situation_sample_problems 讀;格式見那支檔〈證據樣式自測〉)
// === main-ok 標準寫法:兩支互不依賴的呼叫各自 async,最後一起 await
suspend fun load(api: Api): Ui = coroutineScope {
    val a = async { api.a() }
    val b = async { api.b() }
    Ui(a.await(), b.await())
}
// === main-ok 帶 dispatcher 的 async(代碼審 r1 邊界 F3)
suspend fun load(api: Api): Ui = coroutineScope {
    val a = async(Dispatchers.IO) { api.a() }
    val b = async(Dispatchers.IO) { api.b() }
    Ui(a.await(), b.await())
}
// === bad API-14 async 立刻 await,等於沒並行
suspend fun load(api: Api): Ui = coroutineScope {
    val a = async { api.a() }.await()
    val b = async { api.b() }.await()
    Ui(a, b)
}
// === bad API-3 async 的結果沒人接
suspend fun load(api: Api) = coroutineScope {
    async { api.log() }
    api.b()
}
