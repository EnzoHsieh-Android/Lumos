// D-55 的證據樣式自測樣本(規則包自檢 check.py 的 situation_sample_problems 讀;格式見那支檔〈證據樣式自測〉)
// === main-ok 做重活的類別切到注入的 dispatcher
class NewsRepository(private val defaultDispatcher: CoroutineDispatcher = Dispatchers.Default) {
    suspend fun loadNews() = withContext(defaultDispatcher) { parse() }
}
// === main-ok dispatcher 變數名不含 Dispatcher(代碼審 r1 接手 F10 同族)
class Repo(private val io: CoroutineDispatcher = Dispatchers.IO) {
    suspend fun load() = withContext(io) { read() }
}
// === main-ok Composable 裡的計算用 remember 快取
@Composable
fun Contacts(contacts: List<Contact>, comparator: Comparator<Contact>) {
    val sorted = remember(contacts, comparator) { contacts.sortedWith(comparator) }
}
// === bad PERF-6 Composable 裡每次組建都排序
@Composable
fun Contacts(contacts: List<Contact>, comparator: Comparator<Contact>) {
    val sorted = contacts.sortedWith(comparator)
}
// === bad CO-1 類別裡寫死 Dispatchers.Default
class NewsRepository(private val unused: CoroutineDispatcher) {
    suspend fun loadNews() = withContext(Dispatchers.Default) { parse() }
}
