// === main-ok 只改介面宣告:Response<Unit>(代碼審 r1 接手 F2)
interface Api {
    @DELETE("users/{id}") suspend fun delete(@Path("id") id: Long): Response<Unit>
}
// === main-ok 呼叫端用 isSuccessful 判成敗
class Repo(private val api: Api) {
    suspend fun remove(id: Long): Boolean {
        val response = api.delete(id)
        return response.isSuccessful
    }
}
