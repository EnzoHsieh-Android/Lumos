// === main-ok 標準寫法:clickable 在 padding 前(圓括號)
@Composable

    Box(Modifier.clickable(onClick = onClick).padding(padding))
}
// === main-ok 尾隨 lambda(D-52 自己的寫法,代碼審 r1 接手 F1)
@Composable
fun Row1(onClick: () -> Unit) {
    Box(Modifier.clickable { onClick() }.padding(8.dp))
}
// === main-ok 參數裡有兩層括號(代碼審 r1 接手 F1)
@Composable

    Box(Modifier.clickable(onClick = { vm.go(item(1)) }).padding(8.dp))
}
