// === main-ok ViewModel:Pager 的 flow cachedIn
class FeedVm(private val backend: Backend) : ViewModel() {
    val items = Pager(PagingConfig(pageSize = 20)) { FeedSource(backend) }.flow.cachedIn(viewModelScope)
}
// === main-ok 畫面:collectAsLazyPagingItems 並連載入狀態一起看
@Composable
fun Feed(vm: FeedVm) {
    val lazyPagingItems = vm.items.collectAsLazyPagingItems()
    val isLocalEmpty = lazyPagingItems.loadState.refresh is LoadState.NotLoading && lazyPagingItems.itemCount == 0
}
// === alt-ok 有 RemoteMediator:另看 loadState.mediator
@Composable
fun Feed(items: LazyPagingItems<Item>) {
    val syncing = items.loadState.mediator?.refresh is LoadState.Loading
}
// === bad DATA-63 用 itemCount 判空狀態

}
