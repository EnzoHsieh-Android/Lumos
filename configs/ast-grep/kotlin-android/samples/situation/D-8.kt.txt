// D-8 的證據樣式自測樣本(規則包自檢 check.py 的 situation_sample_problems 讀;格式見那支檔〈證據樣式自測〉)
// === main-ok 標準寫法(一行):try 包在 coroutineScope 外面
suspend fun load(api: Api): Ui = try { coroutineScope { val a = async { api.a() }; val b = async { api.b() }; Ui(a.await(), b.await()) } } catch (e: IOException) { Ui.Error }
// === main-ok 標準寫法(多行)
suspend fun load(api: Api): Ui {
    return try {
        coroutineScope {
            val a = async { api.a() }
            val b = async { api.b() }
            Ui(a.await(), b.await())
        }
    } catch (e: IOException) {
        Ui.Error
    }
}
// === bad CO-8 coroutineScope 裡 try 只包 await(代碼審 r1 接手 F4)
suspend fun load(api: Api): Ui = coroutineScope { val d = async { api.fail() }; try { Ui(d.await()) } catch (e: Exception) { Ui.Error } }
// === bad CO-8 多行
suspend fun load(api: Api): Ui = coroutineScope {
    val d = async { api.fail() }
    try {
        Ui(d.await())
    } catch (e: Exception) {
        Ui.Error
    }
}
