// D-22 的證據樣式自測樣本(規則包自檢 check.py 的 situation_sample_problems 讀;格式見那支檔〈證據樣式自測〉)
// === main-ok 複製原請求再改、交給 chain.proceed
class AuthInterceptor(private val token: String) : Interceptor {
    override fun intercept(chain: Interceptor.Chain): Response {
        val request = chain.request().newBuilder().header("Authorization", token).build()
        return chain.proceed(request)
    }
}
// === bad API-20 攔截器吞例外、自造假回應
class SafeInterceptor : Interceptor {
    override fun intercept(chain: Interceptor.Chain): Response {
        return try { chain.proceed(chain.request()) } catch (e: IOException) { Response.Builder().code(599).build() }
    }
}
