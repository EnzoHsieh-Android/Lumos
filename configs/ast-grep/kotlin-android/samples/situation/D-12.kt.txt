// === main-ok 導航點擊用 dropUnlessResumed
@Composable
fun NextButton(onNext: () -> Unit) {
    Button(onClick = dropUnlessResumed { onNext() }) { Text("下一步") }
}
// === main-ok 背景工作用 enqueueUniqueWork(KEEP)
fun schedule(context: Context, req: OneTimeWorkRequest) {
    WorkManager.getInstance(context).enqueueUniqueWork("sync", ExistingWorkPolicy.KEEP, req)
}
// === main-ok 多處同時刷新同一資源:共用 Mutex
class Repo(private val api: Api) {
    private val mutex = Mutex()
    suspend fun refresh() = mutex.withLock { api.refresh() }
}
// === alt-ok 每次都要送:Channel 排隊
class Sender(private val api: Api) {
    private val mutex = Mutex()
    private val outbox = Channel<Msg>(capacity = 64)
    suspend fun send(m: Msg) = outbox.send(m)
}
