// D-52 的證據樣式自測樣本(規則包自檢 check.py 的 situation_sample_problems 讀;格式見那支檔〈證據樣式自測〉)
// === main-ok 可點元素至少 48dp:sizeIn
@Composable
fun Tap(onClick: () -> Unit) {
    Box(Modifier.sizeIn(minWidth = 48.dp, minHeight = 48.dp).clickable { onClick() })
}
// === main-ok 外觀必須小:minimumInteractiveComponentSize 放在 size 之前
@Composable
fun Tap(onClick: () -> Unit) {
    Box(Modifier.clickable { onClick() }.minimumInteractiveComponentSize().size(24.dp))
}
// === main-ok 同一資訊單位合併讀出
@Composable
fun Card1() {
    Row(modifier = Modifier.semantics(mergeDescendants = true) {}) { Text("a"); Text("b") }
}
// === bad UI-106 minimumInteractiveComponentSize 放在 size 之後
@Composable
fun Tap(onClick: () -> Unit) {
    Box(Modifier.size(24.dp).clickable { onClick() }.minimumInteractiveComponentSize())
}
