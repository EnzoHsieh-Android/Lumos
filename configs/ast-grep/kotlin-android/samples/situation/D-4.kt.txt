// D-4 的證據樣式自測樣本(規則包自檢 check.py 的 situation_sample_problems 讀;格式見那支檔〈證據樣式自測〉)
// === main-ok 標準寫法:debounce 之後 flatMapLatest
class SearchVm(private val repo: Repo) : ViewModel() {
    val results = query.debounce(300).flatMapLatest { repo.search(it) }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), emptyList())
}
// === main-ok 只要最新一次的轉換:mapLatest(新值來時取消前一個轉換)
class SearchVm(private val repo: Repo) : ViewModel() {
    val results = query.mapLatest { repo.search(it) }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), emptyList())
}
// === main-ok 命令式:先取消前一個 Job 再 launch
class SearchVm(private val repo: Repo) : ViewModel() {
    private var searchJob: Job? = null
    fun search(q: String) {
        searchJob?.cancel()
        searchJob = viewModelScope.launch { _state.value = repo.search(q) }
    }
}
// === alt-ok 每一次都要做完:collect 依序處理
class UploadVm(private val repo: Repo) : ViewModel() {
    fun start() {
        viewModelScope.launch { uploads.collect { repo.upload(it) } }
    }
}
// === alt-ok 每一次都要做完:Channel 排隊、單一消費者
class UploadVm(private val repo: Repo) : ViewModel() {
    private val queue = Channel<Upload>(capacity = 64)
    fun start() {
        viewModelScope.launch { for (u in queue) repo.upload(u) }
    }
}
// === alt-ok 每一次都要做完:flatMapConcat 依序上傳
class UploadVm(private val repo: Repo) : ViewModel() {
    val done = uploads.flatMapConcat { u -> flow { emit(repo.upload(u)) } }
}
// === alt-ok 每一次都要做完、順序不重要:flatMapMerge 並行上傳
class UploadVm(private val repo: Repo) : ViewModel() {
    val done = uploads.flatMapMerge(concurrency = 4) { u -> flow { emit(repo.upload(u)) } }
}
