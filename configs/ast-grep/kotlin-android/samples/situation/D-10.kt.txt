// D-10 的證據樣式自測樣本(規則包自檢 check.py 的 situation_sample_problems 讀;格式見那支檔〈證據樣式自測〉)
// === main-ok OkHttp 整個呼叫設上限
object Net {
    val client = OkHttpClient.Builder().callTimeout(30, TimeUnit.SECONDS).build()
}
// === main-ok Ktor 裝 HttpTimeout
object Net {
    val client = HttpClient(CIO) { install(HttpTimeout) { requestTimeoutMillis = 30_000 } }
}
// === main-ok 協程 withTimeoutOrNull 接逾時
suspend fun load(api: Api): Ui = withTimeoutOrNull(5_000) { api.load() } ?: Ui.Timeout
// === alt-ok 長連線:不設整體 callTimeout,改 readTimeout
object Net {
    val streaming = OkHttpClient.Builder().readTimeout(60, TimeUnit.SECONDS).build()
}
// === alt-ok 資源在 withTimeout 外取得、用 use
suspend fun read(open: () -> Res): Data = open().use { r -> withTimeout(1_000) { r.read() } }
// === bad CO-14 withTimeout 區塊內取得資源再回傳(代碼審 r1 接手 F5)
suspend fun open(): Res {
    val r = withTimeout(1000) { openResource() }
    return r
}
// === bad API-32 launch 裡 withTimeout 沒處理
fun start(scope: CoroutineScope) {
    scope.launch { withTimeout(5_000) { fetch() } }
}
