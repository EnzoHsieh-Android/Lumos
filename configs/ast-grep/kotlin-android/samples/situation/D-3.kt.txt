// D-3 的證據樣式自測樣本(規則包自檢 check.py 的 situation_sample_problems 讀;格式見那支檔〈證據樣式自測〉)
// === main-ok 標準寫法:map 成 async 再 awaitAll
suspend fun loadAll(api: Api, ids: List<Int>) = coroutineScope {
    ids.map { id -> async { api.load(id) } }.awaitAll()
}
// === main-ok 筆數大時用 Semaphore 限流
suspend fun loadAll(api: Api, ids: List<Int>) = coroutineScope {
    val sem = Semaphore(8)
    ids.map { id -> async { sem.withPermit { api.load(id) } } }.awaitAll()
}
// === main-ok 筆數大時用 limitedParallelism 限流
suspend fun loadAll(api: Api, ids: List<Int>) = coroutineScope {
    val io = Dispatchers.IO.limitedParallelism(8)
    ids.map { id -> async(io) { api.load(id) } }.awaitAll()
}
// === bad API-2 有限流卻逐一 await
suspend fun loadAll(api: Api, ids: List<Int>) = coroutineScope {
    val sem = Semaphore(8)
    ids.map { id -> async { sem.withPermit { api.load(id) } } }.map { it.await() }
}
