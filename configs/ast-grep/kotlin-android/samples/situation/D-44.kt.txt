// D-44 的證據樣式自測樣本(規則包自檢 check.py 的 situation_sample_problems 讀;格式見那支檔〈證據樣式自測〉)
// === main-ok 升版本並給自動遷移
@Database(version = 2, entities = [User::class], autoMigrations = [AutoMigration(from = 1, to = 2)])
abstract class AppDb : RoomDatabase()
// === main-ok 手寫遷移再 addMigrations
fun build(context: Context) = Room.databaseBuilder(context, AppDb::class.java, "app.db").addMigrations(MIGRATION_1_2).build()
// === alt-ok 可以接受清掉資料:fallbackToDestructiveMigration
@Database(version = 3, entities = [User::class], exportSchema = false)
abstract class AppDb : RoomDatabase()
fun build(context: Context) = Room.databaseBuilder(context, AppDb::class.java, "app.db").fallbackToDestructiveMigration(dropAllTables = true).build()
