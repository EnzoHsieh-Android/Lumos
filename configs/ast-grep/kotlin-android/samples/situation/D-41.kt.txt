// D-41 的證據樣式自測樣本(規則包自檢 check.py 的 situation_sample_problems 讀;格式見那支檔〈證據樣式自測〉)
// === main-ok UI 端進行中旗標,先重設再導航
@Composable
fun Form(viewModel: FormVm, onNext: () -> Unit) {
    var validationInProgress by rememberSaveable { mutableStateOf(false) }
    if (validationInProgress) {
        val lifecycle = LocalLifecycleOwner.current.lifecycle
        LaunchedEffect(viewModel, lifecycle) {
            snapshotFlow { viewModel.uiState }.filter { it.isValid }.flowWithLifecycle(lifecycle).collect { validationInProgress = false; onNext() }
        }
    }
}
// === alt-ok 導航由 UI 點擊直接觸發:dropUnlessResumed
@Composable
fun HelpLink(onHelp: () -> Unit) {
    TextButton(onClick = dropUnlessResumed { onHelp() }) { Text("說明") }
}
// === bad ARCH-2 靠狀態觸發導航,沒有 UI 端進行中旗標
@Composable
fun Form(uiState: FormState, navigate: () -> Unit) {
    LaunchedEffect(uiState) { if (uiState.isDone) navigate() }
}
