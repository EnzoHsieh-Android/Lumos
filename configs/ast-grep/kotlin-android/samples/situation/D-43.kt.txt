// D-43 的證據樣式自測樣本(規則包自檢 check.py 的 situation_sample_problems 讀;格式見那支檔〈證據樣式自測〉)
// === main-ok 金鑰放 Android KeyStore
fun keyStore(): KeyStore = KeyStore.getInstance("AndroidKeyStore").apply { load(null) }
// === alt-ok 憑證要跨裝置帶走:改用 Block Store
suspend fun migrate(context: Context) {
    val old = EncryptedSharedPreferences.create(context, "secure", masterKey, scheme, valueScheme)
    Blockstore.getClient(context).storeBytes(StoreBytesData.Builder().setBytes(old.getString("token", "")!!.toByteArray()).build())
}
// === bad DATA-13 權杖照舊放已棄用的加密 prefs
fun prefs(context: Context): SharedPreferences {
    return EncryptedSharedPreferences.create(context, "secure", masterKey, scheme, valueScheme)
}
