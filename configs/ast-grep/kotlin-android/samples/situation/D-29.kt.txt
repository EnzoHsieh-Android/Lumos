// D-29 的證據樣式自測樣本(規則包自檢 check.py 的 situation_sample_problems 讀;格式見那支檔〈證據樣式自測〉)
// === main-ok 要比畫面活得久的工作用注入的外部 scope
class ArticlesRepo(private val externalScope: CoroutineScope, private val ds: ArticlesDs) {
    suspend fun bookmark(article: Article) {
        externalScope.launch { ds.bookmarkArticle(article) }.join()
    }
}
// === main-ok 畫面在生命週期內收集:repeatOnLifecycle
class ListFragment : Fragment() {
    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        viewLifecycleOwner.lifecycleScope.launch {
            viewLifecycleOwner.repeatOnLifecycle(Lifecycle.State.STARTED) { vm.state.collect { render(it) } }
        }
    }
}
// === alt-ok UI 狀態意外變動也沒關係:commitAllowingStateLoss
class HostActivity : AppCompatActivity() {
    fun showBanner(f: Fragment) {
        supportFragmentManager.beginTransaction().replace(R.id.banner, f).commitAllowingStateLoss()
    }
}
// === bad CO-24 長壽 scope 捕捉 Activity

    fun notify(activity: Activity) {
        applicationScope.launch { activity.showToast() }

}
