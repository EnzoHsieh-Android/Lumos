import androidx.compose.ui.Modifier
import androidx.compose.ui.node.ModifierNodeElement

// 壞寫法欄沒有可抄的程式碼:自訂 Element 沒覆寫 inspectableProperties
class MyNode : Modifier.Node()

class MyElement : ModifierNodeElement<MyNode>() {
    override fun create() = MyNode()
    override fun update(node: MyNode) {}
    override fun hashCode(): Int = 1
    override fun equals(other: Any?): Boolean = other is MyElement
}
