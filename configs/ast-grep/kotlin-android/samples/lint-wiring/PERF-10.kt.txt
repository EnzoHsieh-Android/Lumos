import android.content.Context
import android.database.sqlite.SQLiteDatabase
import android.net.Uri
import android.util.AttributeSet
import android.view.VelocityTracker

// obtainStyledAttributes(...)、db.query(...) 沒 use{}、沒 recycle()
fun Context.f(attrs: AttributeSet, db: SQLiteDatabase) {
    VelocityTracker.obtain()
    obtainStyledAttributes(attrs, intArrayOf(android.R.attr.text))
    contentResolver.query(Uri.EMPTY, null, null, null, null)
    db.query("t", null, null, null, null, null, null)
}
