fun f() {
    Class.forName("android.app.ActivityThread").getDeclaredMethod("currentActivityThread")
    Class.forName("android.view.View").getDeclaredMethod("getListenerInfo")
    Class.forName("android.app.Activity").getDeclaredField("mResumed")
    Class.forName("android.os.SystemProperties").getDeclaredMethod("get", String::class.java)
    Class.forName("dalvik.system.VMRuntime").getDeclaredMethod("setHiddenApiExemptions", Array<String>::class.java)
}
