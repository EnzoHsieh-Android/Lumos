import android.net.Uri
import androidx.activity.compose.ManagedActivityResultLauncher
import androidx.compose.runtime.Composable
import androidx.compose.runtime.collectAsState
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.map
import kotlinx.coroutines.launch

class Repo(val flow: Flow<Int>) {
}

suspend fun load() {
}

@Composable fun Screen(scope: CoroutineScope, launcher: ManagedActivityResultLauncher<String, Uri?>, x: String, repo: Repo, init: Int) {
    scope.launch { load() }
    launcher.launch(x)
    repo.flow.map { }.collectAsState(init)
}

@Composable fun CoroutineScope.Inner() {
    launch { }
}
