import android.content.ContentResolver
import android.net.Uri
import android.provider.OpenableColumns
import java.io.File

fun f(cr: ContentResolver, uri: Uri, dir: File): File? {
    cr.query(uri, null, null, null, null)?.use { c ->
        c.moveToFirst()
        val fileName = c.getString(c.getColumnIndexOrThrow(OpenableColumns.DISPLAY_NAME))
        return File(dir, fileName)
    }
    return null
}
