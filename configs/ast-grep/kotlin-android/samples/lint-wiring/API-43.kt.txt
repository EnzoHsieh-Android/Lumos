import android.net.SSLCertificateSocketFactory
import java.net.InetAddress

// 壞寫法欄沒有可抄的程式碼:照文字描述自擬 getInsecure 與 createSocket(InetAddress, …)
fun insecureFactory() = SSLCertificateSocketFactory.getInsecure(0, null)

fun rawSocket(factory: SSLCertificateSocketFactory, addr: InetAddress) = factory.createSocket(addr, 443)
