import android.app.Activity
import android.os.Bundle
import java.io.File

// Activity.onCreate 裡 File.readText()、SharedPreferences.commit()、getSharedPreferences 首次載入
class A1 : Activity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        val text = File(filesDir, "a.txt").readText()
        getSharedPreferences("p", MODE_PRIVATE).edit().putString("k", text).commit()
    }
}
