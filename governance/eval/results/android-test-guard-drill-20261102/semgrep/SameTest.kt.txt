import org.junit.Assert.assertEquals
import com.google.common.truth.Truth
import com.google.common.truth.Truth.assertThat
import com.google.common.truth.Truth.assertWithMessage
import org.hamcrest.MatcherAssert
import org.hamcrest.Matchers.equalTo
import org.junit.Test

class SameTest {
    @Test fun a() {
        val x = calc()
        assertEquals(x, x)
        assertThat(x).isEqualTo(x)
        Truth.assertThat(x).isEqualTo(x)
        assertWithMessage("m").that(x).isEqualTo(x)
        MatcherAssert.assertThat(x, equalTo(x))
        assertThat(x, equalTo(x))
    }
    @Test fun ok() {
        assertThat(calc()).isEqualTo(3)
        assertEquals(3, calc())
        assertThat(calc(), equalTo(3))
    }
}
