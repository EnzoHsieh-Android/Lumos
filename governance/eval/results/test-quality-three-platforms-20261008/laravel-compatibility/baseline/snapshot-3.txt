<?php
namespace Tests\Unit;
use App\Services\Pricing;
use PHPUnit\Framework\TestCase;
final class PricingTest extends TestCase
{
    // Oracle R1 is recorded in REQUIREMENTS.md; expected values are independent literals.
    public function test_amount_100(): void { $this->assertSame(90, (new Pricing)->price(100)); }
    public function test_amount_200(): void { $this->assertSame(180, (new Pricing)->price(200)); }
    public function test_zero(): void { $this->assertSame(0, (new Pricing)->price(0)); }
}
