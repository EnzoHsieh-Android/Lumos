package lab.quality
import android.content.Context
import androidx.test.core.app.ApplicationProvider
import androidx.test.ext.junit.runners.AndroidJUnit4
import org.junit.Before
import org.junit.Test
import org.junit.runner.RunWith
import org.junit.Assert.*
@RunWith(AndroidJUnit4::class)
class QuotePlatformTest {
    private lateinit var context: Context
    @Before fun setUp() {
        context = ApplicationProvider.getApplicationContext()
        assertEquals("lab.quality", context.packageName)
        assertTrue(context.getSharedPreferences("quality_quotes", Context.MODE_PRIVATE).edit().clear().commit())
    }
    @Test fun persistsKnownCharge() {
        QuoteStore(context).save(100)
        assertEquals(90, QuoteStore(context).charged())
    }
    @Test fun rejectsNegativeWithoutPersisting() {
        val store = QuoteStore(context)
        val error = assertThrows(IllegalArgumentException::class.java) { store.save(-1) }
        assertEquals("amount must be nonnegative", error.message)
        assertEquals(-1, store.charged())
    }
}
