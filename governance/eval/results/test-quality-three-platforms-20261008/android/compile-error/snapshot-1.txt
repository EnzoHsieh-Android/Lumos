package lab.quality
import android.content.Context
class QuoteStore(context: Context) {
    private val prefs = context.getSharedPreferences("quality_quotes", Context.MODE_PRIVATE)
    fun save(amount: Int) {
        require(amount >= 0) { "amount must be nonnegative" }
        check(prefs.edit().putInt("charged", Pricing.price(amount)).commit())
    }
    fun charged(): Int = prefs.getInt("charged", -1)
}
