import Foundation
public enum QuoteError: Error { case negativeAmount }
public struct QuoteStore {
    private let defaults: UserDefaults
    public init(defaults: UserDefaults) { self.defaults = defaults }
    public func save(_ amount: Int) throws {
        guard amount >= 0 else { throw QuoteError.negativeAmount }
        defaults.set(Pricing.price(amount), forKey: "charged")
    }
    public var charged: Int? { defaults.object(forKey: "charged") as? Int }
}
