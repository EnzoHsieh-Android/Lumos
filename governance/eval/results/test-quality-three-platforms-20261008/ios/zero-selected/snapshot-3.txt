import XCTest
import Foundation
import QualityCore
final class QuotePlatformTests: XCTestCase {
    private var defaults: UserDefaults?
    private let suite = "lab.quality.test-only"
    override func setUpWithError() throws {
        XCTAssertTrue(ProcessInfo.processInfo.operatingSystemVersion.majorVersion >= 18)
        let storage = try XCTUnwrap(UserDefaults(suiteName: suite))
        storage.removePersistentDomain(forName: suite)
        defaults = storage
    }
    override func tearDownWithError() throws {
        defaults?.removePersistentDomain(forName: suite)
        defaults = nil
    }
    func testPersistsKnownCharge() throws {
        let storage = try XCTUnwrap(defaults)
        try QuoteStore(defaults: storage).save(100)
        XCTAssertEqual(QuoteStore(defaults: storage).charged, 90)
    }
    func testRejectsNegativeWithoutPersisting() throws {
        let store = QuoteStore(defaults: try XCTUnwrap(defaults))
        XCTAssertThrowsError(try store.save(-1)) { error in
            guard case QuoteError.negativeAmount = error else { return XCTFail("wrong refusal reason") }
        }
        XCTAssertNil(store.charged)
    }
}
