import argparse, json, os, subprocess, sys
from pathlib import Path
p=argparse.ArgumentParser();p.add_argument('--zero',action='store_true');a=p.parse_args()
out=Path('.native-results');out.mkdir(exist_ok=True)
dotnet=os.environ.get('LUMOS_LAB_DOTNET','/opt/homebrew/opt/dotnet@9/bin/dotnet')
cmd=[dotnet,'test','--no-restore','--filter','FullyQualifiedName~NoSuchTest' if a.zero else 'FullyQualifiedName~QualityLab.PricingTests', '--logger', 'junit;LogFilePath='+str((out/'report.xml').resolve()), '--logger','trx;LogFileName=native.trx','--results-directory',str(out.resolve())]
r=subprocess.run(cmd,capture_output=True,text=True)
(out/'native.log').write_text(r.stdout+r.stderr)
(out/'native-exit.json').write_text(json.dumps({'argv':cmd,'exit_code':r.returncode},indent=2))
print(r.stdout+r.stderr);print('native_exit_code='+str(r.returncode))
sys.exit(r.returncode)
