using Xunit;
namespace QualityLab;
public class PricingTests
{
    // R1 REQUIREMENTS.md is the oracle, not current implementation output.
    [Fact] public void Amount100() => Assert.Equal(90, Pricing.Price(100));
    [Fact] public void Amount200() => Assert.Equal(180, Pricing.Price(200));
    [Fact] public void Zero() => Assert.Equal(0, Pricing.Price(0));
}
public class CounterexampleTests
{
    [Fact] public void SelfComparison() => Assert.Equal(Pricing.Price(100), Pricing.Price(100));
    [Fact] public void CopiedAlgorithm() { var amount = 100; var expected = amount - amount / 10; Assert.Equal(expected, Pricing.Price(amount)); }
}
