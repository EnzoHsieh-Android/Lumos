<?php
use App\Services\Pricing;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Route;
Route::post('/quotes', function (Request $request, Pricing $pricing) {
    $data = $request->validate(['amount' => 'required|integer|min:0']);
    $charged = $pricing->price((int) $data['amount']);
    DB::table('quotes')->insert(['amount' => $data['amount'], 'charged' => $charged]);
    return response()->json(['charged' => $charged], 201);
});
