<?php
namespace Tests\Unit;
use App\Services\Pricing;
use PHPUnit\Framework\TestCase;
final class CounterexamplesTest extends TestCase
{
    public function test_self_comparison(): void
    {
        $pricing = new Pricing;
        $this->assertSame($pricing->price(100), $pricing->price(100));
    }
    public function test_copied_algorithm(): void
    {
        $amount = 100;
        $expected = $amount - intdiv($amount, 10);
        $this->assertSame($expected, (new Pricing)->price($amount));
    }
}
