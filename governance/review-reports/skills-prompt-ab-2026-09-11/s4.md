```python
text2 = 'diff --git a/scripts/foo.sh b/scripts/sub b/bar.sh\nold mode 100644\nnew mode 100755\n'
print(m._patch_files_from_text(text2))  # {'bar.sh'},非預期的 {'scripts/sub b/bar.sh'}
```
