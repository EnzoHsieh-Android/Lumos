class GoodProvider1 : ContentProvider() { //GOOD
    override fun openFile(uri: Uri, mode: String): ParcelFileDescriptor? {
        val f = File(context!!.filesDir, uri.path)
        if (!f.canonicalPath.startsWith(context!!.filesDir.canonicalPath)) throw SecurityException()
        return ParcelFileDescriptor.open(f, ParcelFileDescriptor.MODE_READ_ONLY)
    }
}
