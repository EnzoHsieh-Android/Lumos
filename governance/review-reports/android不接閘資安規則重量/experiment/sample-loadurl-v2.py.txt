# 外部網址那條第二份語料的排除、同源合併與抽樣(2026-11-08;存成 .txt 免得被當腳本執行)。
# 用法:python3 -I - <語料根目錄> < 本檔(去掉這幾行註解),讀同目錄 hits-loadurl-v2.json,印出抽到的 40 處。
import json, re, hashlib, random, collections, sys, pathlib
C = pathlib.Path(sys.argv[1]); H = sys.argv[2]
TEACH = re.compile(r'(?i)insecure|vulnerab|vuln|goat|damn|ctf|owasp|testdata|sast')
pos = sorted(tuple(x) for x in json.load(open(H)))
kept = []; seen = {}
for f, l in pos:
    if TEACH.search(f): continue                      # 教材與掃描器測試語料不進抽樣
    L = (C / f).read_text(errors='replace').splitlines(); i = l - 1
    key = (f.rsplit('/', 1)[-1], hashlib.sha1(re.sub(r'\s+', '', '\n'.join(L[max(0, i - 20):i + 21])).encode()).hexdigest())
    if key in seen: continue                          # 同檔名、前後 20 行去空白後相同 → 同源,只留第一處
    seen[key] = f; kept.append((f, l))
by = collections.defaultdict(list)
for f, l in kept: by[f.split('/', 1)[0]].append((f, l))
rng = random.Random(20261112); repos = sorted(by); rng.shuffle(repos)
for r in repos[:40]:
    f, l = sorted(by[r])[rng.randrange(len(by[r]))]; print(f"{f}:{l}")
