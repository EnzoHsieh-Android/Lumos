# 抽樣法(2026-11-08 重量用;存成 .txt 免得被當成卷證裡的腳本執行):
# hits-<規則>.json 是 [[語料內路徑, 行], ...];同一個 repo(路徑第一段)的命中歸成一組。
# 命中多的規則:repo 名排序後用 random.Random(20261111) 洗牌,取前 40 個 repo,每個 repo 在自己排序後的命中裡用同一個亂數產生器抽一處。
# 命中 20 多處的規則(Provider、資料庫):全部依路徑、行排序後全判。
import json, random, collections, sys
hits = json.load(open(sys.argv[1]))
by = collections.defaultdict(list)
for f, l in hits:
    by[f.split('/', 1)[0]].append((f, l))
rng = random.Random(20261111)
repos = sorted(by)
rng.shuffle(repos)
for r in repos[:40]:
    f, l = sorted(by[r])[rng.randrange(len(by[r]))]
    print(f"{f}:{l}")
