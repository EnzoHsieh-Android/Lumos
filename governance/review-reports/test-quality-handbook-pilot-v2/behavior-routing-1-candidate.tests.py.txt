import unittest
from subject import route


class RouteTests(unittest.TestCase):
    # 判準來源：需求與已確認例子 route(2,True)=A、route(3,True)=B
    def test_even_user_routes_to_A(self):
        # 輸入 (2, True) -> "A"
        self.assertEqual(route(2, True), "A")

    def test_odd_user_routes_to_B(self):
        # 抓「永遠回 A」：(3, True) -> "B"
        self.assertEqual(route(3, True), "B")

    def test_zero_is_valid_even(self):
        # 規格：非負整數，0 偶數 -> "A"
        self.assertEqual(route(0, True), "A")

    def test_inactive_rejected_with_reason(self):
        # 抓「漏掉 inactive 拒絕」與「理由錯」：合法 user_id 才能到達 active 守衛
        with self.assertRaisesRegex(ValueError, r"^inactive$"):
            route(2, False)
        with self.assertRaisesRegex(ValueError, r"^inactive$"):
            route(3, False)

    def test_negative_user_rejected(self):
        with self.assertRaisesRegex(ValueError, r"^invalid-user$"):
            route(-1, True)

    def test_string_user_rejected(self):
        with self.assertRaisesRegex(ValueError, r"^invalid-user$"):
            route("2", True)

    def test_user_validated_before_active(self):
        # 規格順序：先驗 user_id
        with self.assertRaisesRegex(ValueError, r"^invalid-user$"):
            route(-1, False)
