import unittest
from subject import route


class RouteTests(unittest.TestCase):
    # 判準來源：需求敘述與已確認例子 route(2,True)=A、route(3,True)=B

    def test_even_user_routes_A(self):
        # 已確認例子；守「偶數 → A」
        self.assertEqual(route(2, True), "A")

    def test_odd_user_routes_B(self):
        # 已確認例子；抓「永遠回 A」的 bug
        self.assertEqual(route(3, True), "B")

    def test_zero_is_valid_even(self):
        # 規格：非負整數合法，0 為偶數 → A
        self.assertEqual(route(0, True), "A")

    def test_inactive_valid_user_rejected_with_inactive_reason(self):
        # 抓「漏掉 inactive 拒絕」及「理由錯」；用合法 user_id，確保不是 invalid-user 守衛代打
        with self.assertRaisesRegex(ValueError, r"^inactive$"):
            route(2, False)
        with self.assertRaisesRegex(ValueError, r"^inactive$"):
            route(3, False)

    def test_negative_user_rejected_invalid_user(self):
        with self.assertRaisesRegex(ValueError, r"^invalid-user$"):
            route(-1, True)

    def test_string_user_rejected_invalid_user(self):
        with self.assertRaisesRegex(ValueError, r"^invalid-user$"):
            route("2", True)

    def test_user_check_precedes_active_check(self):
        # 規格：先驗 user_id，再驗 active
        with self.assertRaisesRegex(ValueError, r"^invalid-user$"):
            route(-1, False)
