import unittest
from subject import total


class TotalTests(unittest.TestCase):
    # 判準來源：需求說明中已確認的例子與規則；expected 皆為手算常數，不呼叫被測算法。

    def test_threshold_3_gets_discount_confirmed_example(self):
        # 已確認：total(3,101)=287（303*0.95=287.85 向下取整）
        # 抓：折扣消失（會得 303）、門檻排除 3（>3 會得 303）
        self.assertEqual(total(3, 101), 287)

    def test_above_threshold_discount_floors(self):
        # 4*10=40，95% = 38（整除，手算）；5*7=35 -> 33.25 -> 33
        self.assertEqual(total(4, 10), 38)
        self.assertEqual(total(5, 7), 33)

    def test_below_threshold_full_price_confirmed_example(self):
        # 已確認：total(2,101)=202；抓門檻誤設為 >=2 或一律折扣
        self.assertEqual(total(2, 101), 202)

    def test_zero_quantity_confirmed_example(self):
        self.assertEqual(total(0, 101), 0)

    def test_negative_quantity_reason(self):
        # 抓：錯誤理由回傳其他文字；完整比對訊息
        with self.assertRaisesRegex(ValueError, r"^negative$"):
            total(-1, 101)

    def test_negative_unit_price_reason(self):
        # 單獨測單價為負，數量合法，確認此守衛不由數量守衛代打
        with self.assertRaisesRegex(ValueError, r"^negative$"):
            total(3, -1)
