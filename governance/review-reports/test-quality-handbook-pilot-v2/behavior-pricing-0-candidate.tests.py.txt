import unittest
from subject import total


class TotalTests(unittest.TestCase):
    # 判準來源：使用者已確認例子 total(3,101)=287。
    # 守：quantity=3 落在門檻內、要打 95 折並向下取整（303*0.95=287.85 -> 287）。
    # 應抓到：折扣消失(303)、門檻排除 3(303)、四捨五入而非向下(288)。
    def test_quantity_three_gets_discount_floored(self):
        self.assertEqual(total(3, 101), 287)

    # 判準來源：已確認例子 total(2,101)=202。
    # 守：低於門檻照原價；應抓到門檻誤降為 >=2 或一律打折(191)。
    def test_quantity_two_full_price(self):
        self.assertEqual(total(2, 101), 202)

    # 判準來源：已確認例子 total(0,101)=0。守：零數量邊界。
    def test_quantity_zero_is_zero(self):
        self.assertEqual(total(0, 101), 0)

    # 判準來源：需求「任一負數拋 ValueError("negative")」。
    # 應抓到：錯誤理由回傳其他文字、漏檢某一參數。用 ^...$ 確認完整理由。
    def test_negative_quantity_raises_negative(self):
        with self.assertRaisesRegex(ValueError, r"^negative$"):
            total(-1, 101)

    def test_negative_unit_price_raises_negative(self):
        with self.assertRaisesRegex(ValueError, r"^negative$"):
            total(3, -1)
