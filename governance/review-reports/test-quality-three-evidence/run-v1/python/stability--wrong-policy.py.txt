def price(amount):
    subtotal = amount
    return subtotal - subtotal * 12 // 100
SUBJECT = 'def price(amount):\n    subtotal = amount\n    return subtotal - subtotal * 12 // 100\n'
import json, unittest
results = []
class Tests(unittest.TestCase):
    def test_behavior(self):
        with self.subTest(id='determinism'):
            actual, expected = price(100), price(100)
            results.append({"id": 'determinism', "passed": actual == expected})
            self.assertEqual(actual, expected)
suite = unittest.defaultTestLoader.loadTestsFromTestCase(Tests)
r = unittest.TestResult()
suite.run(r)
print(json.dumps({'tests_run': r.testsRun, 'assertions': results,
 'failures': len(r.failures), 'errors': len(r.errors), 'passed': r.wasSuccessful()}))
