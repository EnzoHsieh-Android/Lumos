def price(amount):
    subtotal = amount
    return subtotal - subtotal // 10
SUBJECT = 'def price(amount):\n    subtotal = amount\n    return subtotal - subtotal // 10\n'
import json, unittest
results = []
class Tests(unittest.TestCase):
    def test_behavior(self):
        with self.subTest(id='amount-100'):
            actual, expected = price(100), 100 - 100 * 12 // 100
            results.append({"id": 'amount-100', "passed": actual == expected})
            self.assertEqual(actual, expected)
        with self.subTest(id='amount-200'):
            actual, expected = price(200), 200 - 200 * 12 // 100
            results.append({"id": 'amount-200', "passed": actual == expected})
            self.assertEqual(actual, expected)
        with self.subTest(id='amount-0'):
            actual, expected = price(0), 0 - 0 * 12 // 100
            results.append({"id": 'amount-0', "passed": actual == expected})
            self.assertEqual(actual, expected)
suite = unittest.defaultTestLoader.loadTestsFromTestCase(Tests)
r = unittest.TestResult()
suite.run(r)
print(json.dumps({'tests_run': r.testsRun, 'assertions': results,
 'failures': len(r.failures), 'errors': len(r.errors), 'passed': r.wasSuccessful()}))
