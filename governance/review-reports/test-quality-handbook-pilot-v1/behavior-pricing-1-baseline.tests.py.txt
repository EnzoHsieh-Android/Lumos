import unittest
from subject import total


class TotalTests(unittest.TestCase):
    # 判準來源：需求給定的已確認例子，expected 為字面值，非由被測算法產生。

    def test_threshold_3_gets_discount_floor(self):
        # 輸入 (3,101)；預期 287（已確認：303*0.95=287.85 向下取整）
        # 抓：折扣消失（會得 303）、門檻排除 3（>3 會得 303）、改四捨五入（288）
        self.assertEqual(total(3, 101), 287)

    def test_below_threshold_full_price(self):
        # 輸入 (2,101)；預期 202（已確認）；抓：門檻放寬成 >=2 或一律打折（會得 191）
        self.assertEqual(total(2, 101), 202)

    def test_zero_quantity(self):
        # 輸入 (0,101)；預期 0（已確認）
        self.assertEqual(total(0, 101), 0)

    def test_negative_quantity_reason(self):
        # 需求：任一負數拋 ValueError("negative")；抓：錯誤理由變成其他文字
        with self.assertRaisesRegex(ValueError, r"^negative$"):
            total(-1, 101)

    def test_negative_unit_price_reason(self):
        # quantity 合法，確保是 unit_price 守衛拒絕，而非 quantity 守衛代打
        with self.assertRaisesRegex(ValueError, r"^negative$"):
            total(3, -1)
