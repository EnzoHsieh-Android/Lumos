import unittest
from subject import route


class RouteTests(unittest.TestCase):
    # 判準來源：需求已確認例子 route(2,True)=A、route(3,True)=B
    # 應抓到：永遠回 A（3 會錯）、固定回 B（2 會錯）
    def test_even_user_routes_to_A(self):
        self.assertEqual(route(2, True), "A")

    def test_odd_user_routes_to_B(self):
        self.assertEqual(route(3, True), "B")

    def test_even_and_odd_land_in_different_groups(self):
        self.assertNotEqual(route(2, True), route(3, True))

    # 判準來源：需求「active 不為 True → ValueError('inactive')」
    # 用合法 user_id 確保能到達 inactive 守衛；應抓到漏拒絕與理由錯
    def test_inactive_even_user_rejected_with_inactive_reason(self):
        with self.assertRaisesRegex(ValueError, r"^inactive$"):
            route(2, False)

    def test_inactive_odd_user_rejected_with_inactive_reason(self):
        with self.assertRaisesRegex(ValueError, r"^inactive$"):
            route(3, False)

    # 判準來源：需求「user_id 非負整數，否則 invalid-user」，且先於 active 檢查
    def test_negative_user_rejected_as_invalid_user(self):
        with self.assertRaisesRegex(ValueError, r"^invalid-user$"):
            route(-1, True)

    def test_non_int_user_rejected_as_invalid_user(self):
        with self.assertRaisesRegex(ValueError, r"^invalid-user$"):
            route("2", True)

    def test_invalid_user_checked_before_active(self):
        with self.assertRaisesRegex(ValueError, r"^invalid-user$"):
            route(-1, False)

    def test_zero_is_valid_even_user(self):
        # 邊界：0 為非負偶數，依規格應回 A
        self.assertEqual(route(0, True), "A")
