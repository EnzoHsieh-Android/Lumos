import unittest
from subject import total


class TotalTests(unittest.TestCase):
    # 守：quantity=3 剛好落在門檻內且有折扣並向下取整。
    # 預期來源：題目已確認例子 total(3,101)=287（303*0.95=287.85→287）。
    # 應抓到：折扣消失（會得 303）、門檻排除 3（會得 303）、改成四捨五入/向上（288）。
    def test_quantity_3_gets_discount_floored(self):
        self.assertEqual(total(3, 101), 287)

    # 守：門檻以下照原價。來源：已確認例子 total(2,101)=202。
    # 應抓到：門檻下移、對所有數量都打折（會得 191）。
    def test_quantity_2_full_price(self):
        self.assertEqual(total(2, 101), 202)

    # 守：數量 0 總價 0。來源：已確認例子 total(0,101)=0。
    def test_quantity_0_is_zero(self):
        self.assertEqual(total(0, 101), 0)

    # 守：負數拋 ValueError 且理由完全是 "negative"。來源：需求。
    # 應抓到：錯誤理由回傳其他文字、漏檢某一參數。
    def test_negative_quantity_rejected_with_reason(self):
        with self.assertRaisesRegex(ValueError, r"^negative$"):
            total(-1, 101)

    def test_negative_unit_price_rejected_with_reason(self):
        with self.assertRaisesRegex(ValueError, r"^negative$"):
            total(3, -1)
