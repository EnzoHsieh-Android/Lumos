import unittest
from subject import total

# 判準來源：需求規格與使用者已確認例子；expected 皆為手算常數，不呼叫被測函式。


class TotalTests(unittest.TestCase):
    # 守：門檻含 3（bug「門檻把 3 排除」）、折扣存在（bug「折扣消失」）、向下取整
    # 3*101=303，303*0.95=287.85 → 287（已確認例子）。若無折扣得 303、四捨五入得 288、門檻 >3 得 303。
    def test_quantity_3_is_discounted_and_floored(self):
        self.assertEqual(total(3, 101), 287)

    # 門檻以上另一點：4*10=40，*0.95=38（手算）。抓「折扣消失」→ 40。
    def test_quantity_above_threshold_discounted(self):
        self.assertEqual(total(4, 10), 38)

    # 門檻以下原價：2*101=202（已確認例子）。抓「門檻錯放到 2」或「一律打折」→ 191。
    def test_quantity_2_full_price(self):
        self.assertEqual(total(2, 101), 202)

    # 0 件：已確認例子 0。
    def test_quantity_0_is_zero(self):
        self.assertEqual(total(0, 101), 0)

    # 錯誤理由必須正是 "negative"（bug「錯誤理由回傳其他文字」），完整比對。
    def test_negative_quantity_reason(self):
        with self.assertRaisesRegex(ValueError, r"^negative$"):
            total(-1, 101)

    # 單價負數也要拒絕；用 quantity=5 確保不是靠 quantity 守衛代打。
    def test_negative_unit_price_reason(self):
        with self.assertRaisesRegex(ValueError, r"^negative$"):
            total(5, -1)
