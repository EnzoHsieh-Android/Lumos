"""Now in Android 正式碼整份當新增切段,量三個數。用法:python3.14 nia_measure.py <repo> <型錄檔> <NiA 根>"""
import importlib.machinery as M, importlib.util as U, sys
from pathlib import Path
sys.dont_write_bytecode = True
repo, cat, nia = Path(sys.argv[1]), Path(sys.argv[2]), Path(sys.argv[3])
def load(p, n):
    l = M.SourceFileLoader(n, str(p)); s = U.spec_from_loader(n, l); m = U.module_from_spec(s); l.exec_module(m); return m
kp = load(repo / "configs/ast-grep/kotlin-android/check.py", "kp"); lm = load(repo / "scripts/lumos", "lm")
rows, _ = kp.dt_table(cat.read_text(encoding="utf-8"))
files = hit = segs = pairs = 0
for p in sorted(nia.rglob("*.kt")):
    rel = str(p.relative_to(nia))
    if "/build/" in rel or not lm._kt_sit_path_ok(rel):
        continue
    files += 1
    ss = lm._kt_sit_text_segments(rel, p.read_text(encoding="utf-8"), rows)
    hit += bool(ss); segs += len(ss); pairs += sum(len(s["rows"]) for s in ss)
print(f"正式碼檔 {files},至少命中一列 {hit}({hit / files:.1%}),情境段 {segs},(段,列)配對 {pairs}")
