#!/usr/bin/env python3
"""Freeze three historical weak/strong test controls, without model calls."""
import argparse
import json
from pathlib import Path
import subprocess
import tempfile

import historical_test_quality as h

JAVA_BASE = '631d7e3bc93f6f0ce76eb5a25fbaa6360c50a2f9'
JAVA_FIXED = '2e2618464ef35e3b6740c44799f986c4cbea89cc'
JAVA_PATCH = 'governance/review-reports/code-java補棧/r1-snapshot.patch'
JAVA_PATCH_SHA = '5ef6bda8dcd87e9e230b155020a3c73db0478e39d0d23eb9da9456371bed0c09'
JAVA_BLOBS = {'scripts/lumos': '29cd6837dcab8aff885186b021032b83d2b4e039',
              'scripts/test_lumos.py': '324c35a186a5e3919cb83986a2205780f8f5592e'}
CASES = [
    {'id': 'lock-reentrancy', 'role': 'grader-control', 'shape': 'claimed-scenario-not-established',
     'method': h.METHOD, 'weak_checks': 4, 'strong_checks': 4, 'failed_indices': [3]},
    {'id': 'java-discovery', 'role': 'effect-candidate', 'shape': 'missing-counterexamples',
     'method': 't_java_profile_discovery', 'weak_checks': 8, 'strong_checks': 12, 'failed_indices': [8, 9, 10, 11]},
    {'id': 'fence-recovery', 'role': 'effect-candidate', 'shape': 'fix-broader-than-covered-inputs',
     'method': 't_closing_fence_must_not_carry_a_language_tag', 'weak_checks': 5, 'strong_checks': 8, 'failed_indices': [5, 6]},
]


def commit_files(commit):
    return {file: h.historical_file(commit, file) for file in JAVA_BLOBS}


def java_snapshot():
    patch = h.REPO / JAVA_PATCH
    if h.sha(patch.read_text()) != JAVA_PATCH_SHA:
        raise ValueError('historical-patch-sha-mismatch')
    with tempfile.TemporaryDirectory(prefix='history-java-snapshot-') as directory:
        work = Path(directory)
        subprocess.run(['git', 'init', '-q', str(work)], check=True, timeout=10)
        (work / 'scripts').mkdir()
        for file, text in commit_files(JAVA_BASE).items():
            (work / file).write_text(text)
        (work / 'scripts/lumos').chmod(0o755)
        subprocess.run(['git', 'apply', '--include=scripts/lumos', '--include=scripts/test_lumos.py', str(patch)],
                       cwd=work, capture_output=True, check=True, timeout=20)
        result = {file: (work / file).read_text() for file in JAVA_BLOBS}
        for file, text in result.items():
            blob = subprocess.run(['git', 'hash-object', '--stdin'], input=text, capture_output=True,
                                  text=True, check=True, timeout=10).stdout.strip()
            if blob != JAVA_BLOBS[file]:
                raise ValueError('reconstructed-blob-mismatch:' + file)
        return result


def load_materials():
    faulty, fixed = (commit_files(h.VERSIONS[key]) for key in ('faulty', 'fixed'))
    java = java_snapshot()
    return {
        'lock-reentrancy': {'faulty': faulty, 'fixed': fixed,
                            'provenance': {'type': 'commits', **h.VERSIONS}},
        'fence-recovery': {'faulty': faulty, 'fixed': fixed,
                           'provenance': {'type': 'commits', **h.VERSIONS}},
        'java-discovery': {'faulty': java, 'fixed': commit_files(JAVA_FIXED),
                           'provenance': {'type': 'snapshot-and-commit', 'base': JAVA_BASE,
                                          'snapshot': JAVA_PATCH, 'snapshot_sha256': JAVA_PATCH_SHA,
                                          'reconstructed_blobs': JAVA_BLOBS, 'fixed': JAVA_FIXED}},
    }


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--out', type=Path, required=True)
    args = parser.parse_args()
    args.out.mkdir(parents=True, exist_ok=False)
    materials = load_materials()
    manifest = {'purpose': 'historical-control-qualification-not-model-effect', 'cases': CASES,
                'runner_sha256': h.sha(Path(__file__).read_text()),
                'replay_dependency_sha256': h.sha(Path(h.__file__).read_text()),
                'provenance': {key: value['provenance'] for key, value in materials.items()},
                'file_sha256': {key: {version: {f: h.sha(text) for f, text in value[version].items()}
                                     for version in ('faulty', 'fixed')} for key, value in materials.items()}}
    # Preset cases/counts/targets are saved before any replay; no score-guided selection.
    (args.out / 'manifest.json').write_text(json.dumps(manifest, ensure_ascii=False, indent=2) + '\n')
    rows, tests = [], {}
    for case in CASES:
        material = materials[case['id']]
        original = {kind: h.test_function(material[version]['scripts/test_lumos.py'], case['method'])
                    for kind, version in [('weak', 'faulty'), ('strong', 'fixed')]}
        tests[case['id']] = {'original': original, 'executed': dict(original)}
        if case['id'] == 'lock-reentrancy':
            tests[case['id']]['executed']['strong'] = original['strong'].replace('_sig.alarm(8)', '_sig.alarm(2)', 1)
        for version in ('faulty', 'fixed'):
            for kind, code in tests[case['id']]['executed'].items():
                row = h.replay(material[version]['scripts/lumos'], code, method=case['method'],
                               require_precondition_label=case['id'] != 'java-discovery')
                row.update(case=case['id'], version=version, kind=kind, test_sha256=h.sha(code))
                checks = row.get('checks', [])
                failed = [i for i, c in enumerate(checks) if not c['passed']]
                expected = case['failed_indices'] if version == 'faulty' and kind == 'strong' else []
                row['qualified'] = row['status'] == 'executed' and len(checks) == case[kind + '_checks'] and failed == expected
                rows.append(row)
                print(case['id'], version, kind, 'qualified=' + str(row['qualified']), 'failed=' + str(failed), flush=True)
                (args.out / 'results.json').write_text(json.dumps({'rows': rows}, ensure_ascii=False, indent=2) + '\n')
    (args.out / 'test-materials.json').write_text(json.dumps(tests, ensure_ascii=False, indent=2) + '\n')
    passed = all(row['qualified'] for row in rows)
    (args.out / 'summary.json').write_text(json.dumps({'controls_passed': passed, 'cells': len(rows),
        'case_count': len(CASES), 'effect_candidates': [c['id'] for c in CASES if c['role'] == 'effect-candidate'],
        'retired_effect_control': 'lock-reentrancy', 'model_calls': 0}, indent=2) + '\n')
    return 0 if passed else 2


if __name__ == '__main__':
    raise SystemExit(main())
