用法:python3 -I sweep.py [--list]"""

def blocks(path):
    """回 (起行, 迄行) 的區塊,只取 IDS 或 utils 檔整份"""
    L=open(path,encoding='utf-8').read().split('\n')
    if 'utils/' in path: return [(0,len(L))]
    out=[]; start=0
    for i,l in enumerate(L+['---']):
        if l=='---' or i==len(L):
            seg=L[start:i]
            if any(re.match(r'^id: (\S+)',x) and re.match(r'^id: (\S+)',x).group(1) in IDS for x in seg): out.append((start,i))
            start=i+1
    return out
def alts(rx):
    """正規式裡每個含 | 的括號群組,逐一列出拿掉某個選項後的字串"""
    res=[]
    for i,ch in enumerate(rx):
        if ch!='(' or (i>0 and rx[i-1]=='\\'): continue
        depth=0; parts=[]; last=i+1; j=i
        while j<len(rx):
            c=rx[j]
            if c=='\\': j+=2; continue
            if c=='(': depth+=1
            elif c==')':
                depth-=1
                if depth==0: parts.append((last,j)); break
            elif c=='|' and depth==1: parts.append((last,j)); last=j+1
            j+=1
        if len(parts)>=2:
            for a,b in parts:
                body=rx[a:b]
                if body.startswith('?:') or body.startswith('?'): continue
                keep=[rx[x:y] for x,y in parts if (x,y)!=(a,b)]
                res.append((body, rx[:i+1]+'|'.join(keep)+rx[parts[-1][1]:]))
    return res
muts=[]
for f in FILES:
    L=open(f,encoding='utf-8').read().split('\n')
    for s,e in blocks(f):
        for n in range(s,e):
            l=L[n]
            if re.match(r'^\s*- pattern: ',l):
                muts.append((f,n,l,None,f'拿掉樣式 {l.strip()[11:60]}'))
            m=re.search(r"regex: '(.*)'",l) or re.search(r'regex: (\^.*\$)\s*$',l)
            if m:
                for body,new in alts(m.group(1)):
                    muts.append((f,n,l,l.replace(m.group(1),new,1),f'正規式拿掉選項 {body[:40]}'))
if '--list' in sys.argv:
