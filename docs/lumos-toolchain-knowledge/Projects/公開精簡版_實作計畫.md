# 公開精簡版 實作計畫

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development`（建議）或 `superpowers:executing-plans` 逐 Task 實作。步驟用 `- [ ]` 追蹤。

> **計畫落點說明**：writing-plans 預設寫 `docs/superpowers/plans/`，但本 repo 的 CLAUDE.md 規定「計劃/設計一律寫成圖譜 `Projects/` 節點、不寫其他 repo 路徑」，故落此處（skill 本身允許使用者偏好覆蓋預設）。

**Goal:** 從既有單檔 CLI 生成一個「只留 24 支讀寫維護指令」的精簡版，配一支只動 `$HOME` 的安裝器與一份自足 README，一次性交給離職接手者。

**Architecture:** 三支新腳本留在本 repo 當**工具**（生成器／掃描器／安裝器範本），跑生成後產物落 `dist/`（gitignore）；`dist/` 通過四道驗證後由使用者 push 進一個**凍結的新 repo**。移除謂詞＝「保留指令 dispatch 的 AST 可達性閉包之補集」，不是 `cmd_` 前綴比對。

**Tech Stack:** Python 3 stdlib（`ast`／`argparse`／`subprocess`／`pathlib`）、bash（安裝器）、既有 `scripts/test_lumos.py` 測試框架。

## Global Constraints

- **零依賴家規**：只准 Python 3 stdlib 與 bash，不得 `pip install` 任何東西。
- **測試位置與跑法**：所有測試寫進 `scripts/test_lumos.py`，函式名以 `t_slim_` 開頭（`main()` 用 `globals()` 掃 `t_` 前綴自動收集）。跑法 `python3 scripts/test_lumos.py`，或 `python3 scripts/test_lumos.py -k slim` 只跑本批。**注意：`-k` 選中 0 個案例會刻意回 rc1**。
- **測試 helper**：`run(vault, *args, expect_rc=None)` 跑 lumos 子指令；`check(name, cond, detail="")` 記一條斷言；`mkvault()` 建私有 vault。**每個測試自建 `tempfile.mkdtemp()`**（既有慣例，避免帳檔互踩）。
- **★不動 Landmark★**：`/Users/enzo/backend/LandmarkMember` 全程唯讀，不得寫入。
- **保留 24 支**：`append archive backlinks context contracts decision-add decision-reindex decision-supersede decisions doctor export guard links lint map new recent rel-cascade search set show stale stats sync-verified-by`
- **移除 29 支**：`anchor bootstrap canary ci-status ci-wait cochange code-loop compose-metrics deinit fold-check gov impact init install lint-check lint-watch loop pitfalls refcheck self-audit signoff spec-trace sqlfluff-sarif stylelint-sarif teardown test-layers testmap uninstall update`
- **指令全集真值一律取 `python3 scripts/lumos --help` 的 choices，不得硬編 53 這個數字**。
- 產物落 `dist/`，**不進版控**。

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""slim-scan.py — 交付文字懸空引用掃描器(stdlib only)

掃 README/SKILL.md/reference.md,找出提及「精簡版已移除的指令」或「不交付的 skill」
的句子,輸出候選清單交人逐條裁。

★不是自動改寫器★:裸 token 形態必然有假陽性(export/set/show/loop/impact 等本身
是常見英文詞),故只出候選、不動檔案。

rc: 0=無候選 / 1=有候選 / 2=參數錯
"""
import argparse
import json
import re
import subprocess
import sys
from pathlib import Path

# 不交付的 skill(含不帶 lumos- 前綴的簡稱)
DROP_SKILLS = ("lumos-design-loop", "lumos-code-loop", "lumos-core-knowledge",
               "lumos-pitfalls-gapfill", "design-loop", "code-loop",
               "core-knowledge", "pitfalls-gapfill")

KEEP = set("""append archive backlinks context contracts decision-add decision-reindex
decision-supersede decisions doctor export guard links lint map new recent
rel-cascade search set show stale stats sync-verified-by""".split())


def all_commands(lumos: Path):
    """真值取 --help 的 choices,不硬編。"""
    r = subprocess.run([sys.executable, str(lumos), "--help"],
                       capture_output=True, text=True)
    m = re.search(r"\{([a-z0-9,\-]+)\}", r.stdout)
    if not m:
        print("ERROR: 無法從 --help 解析指令全集", file=sys.stderr)
        sys.exit(2)
    return set(m.group(1).split(","))


def scan_line(line, removed):
    """回傳 [(token, form), ...]。同一行可命中多種形態。"""
    hits = []
    # 形態 1:帶前綴
    for m in re.finditer(r"(?:python3\s+)?(?:scripts/)?lumos\s+([a-z0-9\-]+)", line):
        if m.group(1) in removed:
            hits.append((m.group(1), "prefixed"))
    # 形態 2/4:backtick span —— 取首 token 比對(涵蓋「裸 token」與「帶參數」兩種)
    for m in re.finditer(r"`([^`]+)`", line):
        span = m.group(1).strip()
        first = span.split()[0] if span.split() else ""
        first = first.lstrip("/")
        if first in removed:
            hits.append((first, "bare-token" if span == first else "span-with-args"))
    # 形態 3:skill 名(含簡稱)
    for s in DROP_SKILLS:
        if s in line:
            hits.append((s, "skill-name"))
            break
    # 形態 5:裸散文型 —— 無 backtick、無前綴,直接嵌在句子裡
    #   只認「有明確詞邊界」的出現,且該 token 不是保留指令
    for cmd in removed:
        if len(cmd) < 4:          # 太短的詞(如 gov)誤報率過高,交給形態 1/2
            continue
        if re.search(r"(?<![`\w\-])" + re.escape(cmd) + r"(?![\w\-])", line):
            if not any(h[0] == cmd for h in hits):
                hits.append((cmd, "prose"))
    return hits


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("files", nargs="+")
    ap.add_argument("--json", action="store_true")
    ap.add_argument("--lumos", default=str(Path(__file__).resolve().parent / "lumos"))
    a = ap.parse_args()

    removed = all_commands(Path(a.lumos)) - KEEP
    out = []
    for fp in a.files:
        p = Path(fp)
        if not p.is_file():
            print(f"ERROR: 檔案不存在: {fp}", file=sys.stderr)
            return 2
        for n, line in enumerate(p.read_text(encoding="utf-8").splitlines(), 1):
            for tok, form in scan_line(line, removed):
                out.append({"file": fp, "line": n, "token": tok,
                            "form": form, "text": line.strip()[:120]})

    if a.json:
        print(json.dumps({"candidates": out, "total": len(out)}, ensure_ascii=False))
    else:
        for c in out:
            print(f"{c['file']}:{c['line']}  [{c['form']}] {c['token']}\n    {c['text']}")
        print(f"\n候選 {len(out)} 條 —— ★這是候選不是判決,裸 token 與散文型必有假陽性,請逐條裁★")
    return 1 if out else 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""slim-gen.py — 從單檔 CLI 生成「只留白名單指令」的精簡版(stdlib only)

★移除謂詞不是「cmd_ 前綴」★(lint-watch 實作叫 _lint_watch_mode、保留的 doctor
叫 run_doctor)——正解是「保留指令 dispatch 的 AST 可達性閉包之補集」。

★root 集合必須含 module-level 語句與 class body★,否則 `_SKILLS = _skills_list()`
這種模組層賦值會讓 helper 落進補集,產物 import 期就 NameError 全滅。
"""
import argparse
import ast
import re
import sys
from pathlib import Path

DEFAULT_KEEP = """append archive backlinks context contracts decision-add decision-reindex
decision-supersede decisions doctor export guard links lint map new recent
rel-cascade search set show stale stats sync-verified-by""".split()


def _called(node, universe):
    out = set()
    for c in ast.walk(node):
        if isinstance(c, ast.Call) and isinstance(c.func, ast.Name):
            out.add(c.func.id)
        elif isinstance(c, ast.Name) and isinstance(c.ctx, ast.Load) and c.id in universe:
            out.add(c.id)            # 函式當一級物件傳遞
    return out & universe


def _cmd_names(test):
    return {n.value for n in ast.walk(test)
            if isinstance(n, ast.Constant) and isinstance(n.value, str)}


def build(src_text, keep):
    tree = ast.parse(src_text)
    funcs = {n.name: n for n in ast.walk(tree) if isinstance(n, ast.FunctionDef)}
    main = funcs.get("main")
    if main is None:
        raise SystemExit("ERROR: 找不到 main()")

    roots = set()
    # root ①:保留指令的 dispatch 分支
    for node in ast.walk(main):
        if isinstance(node, ast.If):
            ns = _cmd_names(node.test)
            if ns and (ns & set(keep)):
                for st in node.body:
                    roots |= _called(st, set(funcs))
    # root ②:module-level 語句與 class body(★不可省★)
    for node in tree.body:
        if not isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            roots |= _called(node, set(funcs))
    roots |= _called(main, set(funcs)) & {"main"}
    roots.add("main")

    seen, stack = set(roots), list(roots)
    while stack:
        f = stack.pop()
        for g in _called(funcs[f], set(funcs)):
            if g not in seen:
                seen.add(g); stack.append(g)
    drop = set(funcs) - seen
    return tree, funcs, seen, drop


class _Pruner(ast.NodeTransformer):
    def __init__(self, drop, keep_cmds):
        self.drop, self.keep = drop, set(keep_cmds)

    def visit_FunctionDef(self, node):
        if node.name in self.drop:
            return None
        self.generic_visit(node)
        return node

    def visit_If(self, node):
        ns = _cmd_names(node.test)
        if ns and not (ns & self.keep) and any(re.fullmatch(r"[a-z0-9\-]+", n) for n in ns):
            return None                      # 純移除指令的 dispatch 分支
        self.generic_visit(node)
        return node

    def visit_Expr(self, node):
        # 砍 sub.add_parser("<removed>", ...) 的 Expr 形式註冊
        c = node.value
        if (isinstance(c, ast.Call) and isinstance(c.func, ast.Attribute)
                and c.func.attr == "add_parser" and c.args
                and isinstance(c.args[0], ast.Constant)
                and isinstance(c.args[0].value, str)
                and re.fullmatch(r"[a-z0-9\-]+", c.args[0].value)
                and c.args[0].value not in self.keep):
            return None
        return node

    def visit_Assign(self, node):
        c = node.value
        if (isinstance(c, ast.Call) and isinstance(c.func, ast.Attribute)
                and c.func.attr == "add_parser" and c.args
                and isinstance(c.args[0], ast.Constant)
                and isinstance(c.args[0].value, str)
                and re.fullmatch(r"[a-z0-9\-]+", c.args[0].value)
                and c.args[0].value not in self.keep):
            return None
        return node

    def visit_For(self, node):
        """迴圈註冊:過濾迭代元素,全空則整段砍。"""
        it = node.iter
        if isinstance(it, ast.Tuple) and any(
                isinstance(e, ast.Tuple) and e.elts
                and isinstance(e.elts[0], ast.Constant) for e in it.elts):
            kept = [e for e in it.elts
                    if not (isinstance(e, ast.Tuple) and e.elts
                            and isinstance(e.elts[0], ast.Constant)
                            and e.elts[0].value not in self.keep)]
            if not kept:
                return None
            node.iter = ast.Tuple(elts=kept, ctx=ast.Load())
        self.generic_visit(node)
        return node


def main():
    ap = argparse.ArgumentParser()
    here = Path(__file__).resolve().parent
    ap.add_argument("--src", default=str(here / "lumos"))
```
