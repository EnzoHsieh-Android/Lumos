## 回放腳本

在主線副本上跑:`python3.14 replay_repair_size.py <repo 根>`,印出上面每一個數字。

```python
"""代碼審續輪回放:修補新增行數與下一輪 major 的關係,以及發現落在修補差異內外。

用法:python3.14 replay_repair_size.py <repo 根>
讀 <repo>/docs/.canary-log.jsonl(分級、每輪折入數)與 <repo>/governance/review-reports/code-*/rN-snapshot.patch、rN-*.md 席報告。
口徑(與修復穩定性試行 S6 同一單位):
- 只取分級標成 standard 或 high、名字以 code- 開頭、凍結材料是 patch 的迴圈;第 N 輪(N>=2)對比第 N-1 輪。
- 修補新增行數 = 第 N 輪凍結 patch 裡以 + 開頭(不含 +++ 檔頭)的行,扣掉第 N-1 輪已有的同內容行(逐行序列比對);
  路徑在 governance/review-reports/、governance/replay/ 或 docs/.*-log.jsonl 的檔不算。超過 5000 行(重定基底)的輪排除。
- 發現落點:每條發現(獨立 severity: 行之後那段)的引句去空白後,只在修補差異行出現=差異內,只在沒變的行=差異外。
- major = 該條發現的 severity 是 major 或 blocker。
"""
import collections, difflib, json, re, statistics, sys
from pathlib import Path

ROOT = Path(sys.argv[1])
SEV = ["clean", "minor", "major", "blocker"]
Q = re.compile(r"引句[^「\n]{0,20}「(.+?)」")
SKIP = re.compile(r"^(governance/review-reports/|governance/replay/|docs/\.[^/]*-log\.jsonl$)")


def norm(s):
    return re.sub(r"\s+", "", s.replace("`", "").replace("**", ""))


def patch_lines(p):
    """回 (加的行, 全部內容行),跳過不算的檔。"""
    added, body, skip = [], [], False
    for ln in p.read_text(encoding="utf-8", errors="replace").splitlines():
        m = re.match(r"^diff --git a/(\S+) b/(\S+)", ln)
        if m:
            skip = bool(SKIP.match(m.group(2)))
            continue
        if skip or ln.startswith(("+++", "---", "@@", "index ")):
            continue
        if ln[:1] in "+- ":
            body.append(ln[1:])
            if ln[:1] == "+":
                added.append(ln[1:])
    return added, body


def findings(rep):
    out, cur, first = [], None, True
    for ln in rep.read_text(encoding="utf-8", errors="replace").splitlines():
        m = re.match(r"^severity:\s*(\w+)", ln)
        if m:
            if first:
                first = False
                continue
            cur = {"sev": m.group(1).lower(), "quotes": []}
            out.append(cur)
            continue
        if cur is not None:
            cur["quotes"] += Q.findall(ln)
    return out


rows = [json.loads(l) for l in (ROOT / "docs/.canary-log.jsonl").open(encoding="utf-8") if l.strip()]
tier, fold = {}, collections.Counter()
for r in rows:
    lp = r.get("loop")
    if not lp or r.get("kind") == "spec-gate":
        continue
    if r.get("tier"):
        tier.setdefault(lp, r["tier"])
    if r.get("findings_set") and re.fullmatch(r"r\d+", r.get("round") or ""):
        fold[(lp, int(r["round"][1:]))] += len(r.get("folded_set") or []) + len(r.get("accepted_set") or [])

data, where = [], collections.Counter()
for d in sorted((ROOT / "governance/review-reports").glob("code-*")):
    if tier.get(d.name) not in ("standard", "high"):
        continue
    sn = {int(m.group(1)): p for p in d.glob("r*-snapshot.patch") if (m := re.match(r"r(\d+)-snapshot\.patch$", p.name))}
    for n in sorted(sn):
        if n < 2 or n - 1 not in sn:
            continue
        pa, pb = patch_lines(sn[n - 1])
        ca, cb = patch_lines(sn[n])
        new_added = []
        for tag, i1, i2, j1, j2 in difflib.SequenceMatcher(None, pa, ca, autojunk=False).get_opcodes():
            if tag in ("replace", "insert"):
                new_added += ca[j1:j2]
        if len(new_added) > 5000:
            continue
        newb, oldb = [], []
        for tag, i1, i2, j1, j2 in difflib.SequenceMatcher(None, pb, cb, autojunk=False).get_opcodes():
            (oldb if tag == "equal" else newb).extend(cb[j1:j2])
        newn, oldn = norm("\n".join(newb)), norm("\n".join(oldb))
        majors = majors_all = 0
        for rep in d.glob(f"r{n}-*.md"):
            if re.match(rf"r{n}-(snapshot|intake|carrier|repair|fix|dispatch)", rep.name):
                continue
            for f in findings(rep):
                qs = [norm(q) for q in f["quotes"] if len(norm(q)) >= 6]
                big = f["sev"] in ("major", "blocker")
                if not qs:
                    w = "沒引句"
                elif any(q in newn and q not in oldn for q in qs):
                    w = "差異內"
                elif any(q in newn for q in qs):
                    w = "新舊都有"
                elif any(q in oldn for q in qs):
                    w = "差異外"
                else:
                    w = "錨不到"
                where[(w, big)] += 1
                if big and w in ("差異內", "新舊都有"):
                    majors += 1
                if big:
                    majors_all += 1
        data.append({"loop": d.name, "round": n, "added": len(new_added), "majors": majors, "majors_all": majors_all, "prev_fold": fold.get((d.name, n - 1))})

rounds_of = collections.defaultdict(set)
sev_of = collections.defaultdict(int)
for r in rows:
    lp, rd = r.get("loop"), r.get("round") or ""
    if lp and lp.startswith("code-") and tier.get(lp) in ("standard", "high") and re.fullmatch(r"r\d+", rd) and r.get("kind") != "spec-gate":
        rounds_of[lp].add(int(rd[1:]))
        sev_of[(lp, int(rd[1:]))] = max(sev_of[(lp, int(rd[1:]))], SEV.index(r.get("severity") or "clean") if (r.get("severity") or "clean") in SEV else 0)
multi = [lp for lp, rs in rounds_of.items() if len(rs) >= 2]
capped = [lp for lp in multi if max(rounds_of[lp]) >= 3]
r3maj = [lp for lp in capped if sev_of[(lp, 3)] >= 2]
print(f"迴圈:standard/high 代碼審跑兩輪以上 {len(multi)} 個,跑滿三輪 {len(capped)}({len(capped) / len(multi):.0%}),其中第三輪帳上最高仍 ≥major {len(r3maj)}")
withmaj = [x for x in data if x["majors_all"]]
print(f"續輪有 major 的 {len(withmaj)} 輪裡,只算修補差異內就沒有 major 的 {sum(1 for x in withmaj if not x['majors'])} 輪")
print(f"續輪 {len(data)} 輪(standard/high 代碼審,排除新增超過 5000 行)")
tot = sum(v for (w, b), v in where.items() if w in ("差異內", "新舊都有", "差異外"))
out = sum(v for (w, b), v in where.items() if w == "差異外")
print(f"錨得到的發現 {tot},落在差異外 {out}({out / tot:.0%})" if tot else "沒有錨得到的發現")
data.sort(key=lambda x: x["added"])
q = len(data) // 4
print("修補新增行數四分組:")
for i in range(4):
    part = data[i * q:(i + 1) * q] if i < 3 else data[3 * q:]
    hit = sum(1 for x in part if x["majors"])
    print(f"  {part[0]['added']}–{part[-1]['added']} 行:{len(part)} 輪,下一輪(差異內)有 major {hit}({hit / len(part):.0%}),平均 {statistics.mean(x['majors'] for x in part):.2f}")
for lim in (100,):
    s = [x for x in data if x["added"] <= lim]
    print(f"  新增 ≤{lim} 行:{len(s)} 輪,有 major {sum(1 for x in s if x['majors'])}({sum(1 for x in s if x['majors']) / max(1, len(s)):.0%})")
byl = collections.defaultdict(list)
for x in data:
    byl[x["loop"]].append(x)
small, big = [], []
for xs in byl.values():
    if len(xs) < 2:
        continue
    med = statistics.median(x["added"] for x in xs)
    for x in xs:
        if x["added"] < med:
            small.append(x["majors"])
        elif x["added"] > med:
            big.append(x["majors"])
print(f"同一迴圈內:比自己中位數小 {len(small)} 輪,有 major {sum(1 for v in small if v)};大 {len(big)} 輪,有 major {sum(1 for v in big if v)}")
st = [x for x in data if x["prev_fold"]]
fs = sorted(x["prev_fold"] for x in st)
cuts = [fs[len(fs) // 3], fs[2 * len(fs) // 3]]
for name, sel in (("少", lambda f: f <= cuts[0]), ("中", lambda f: cuts[0] < f <= cuts[1]), ("多", lambda f: f > cuts[1])):
    part = [x for x in st if sel(x["prev_fold"])]
    m = statistics.median(x["added"] for x in part)
    s = [x["majors"] for x in part if x["added"] <= m]
    b = [x["majors"] for x in part if x["added"] > m]
    print(f"上一輪折入{name}({min(x['prev_fold'] for x in part)}–{max(x['prev_fold'] for x in part)} 條,{len(part)} 輪):"
          f"修補小的一半有 major {sum(1 for v in s if v)}/{len(s)},大的一半 {sum(1 for v in b if v)}/{len(b)}")
```
